    function extractPR(text) {

      const match = text.toUpperCase().match(/PR\s?\d{4}/);

      return match ? match[0].replace(' ', '') : null;

    }

    function readyProjects() {

      const ready = Object.entries(projects).filter(([_, p]) => p.readiness >= 85);

      if (!ready.length) return "No projects are fully ready for execution.";

      return `Projects currently ready or near-ready for execution:<br /><br />` + ready.map(([pr, p]) => `<strong>${pr}</strong> – ${p.name}: ${p.readiness}% ready, ${p.status}`).join('<br />');

    }


    function answer(text) {

      const pr = extractPR(text);

      const lower = text.toLowerCase();


      if (lower.includes('ready projects') || lower.includes('ready for execution')) {

        return readyProjects();

      }


      if (!pr) {

        return "Please provide a PR number so I can check the project status. Example: <strong>PR9160</strong>.";

      }


      const p = projects[pr];

      if (!p) {

        return `I could not find <strong>${pr}</strong> in the demo project register. Please check the PR number or contact the IHP project manager.`;

      }


      if (lower.includes('delay') || lower.includes('delayed')) {

        return `<strong>${pr}</strong> is delayed mainly because of the following open items:<br /><br /><ul>${p.pending.map(x => `<li>${x}</li>`).join('')}</ul><br />Recommended next action: ${p.next}`;

      }


      if (lower.includes('approval') || lower.includes('pending')) {

        return `<strong>Pending approvals / blockers for ${pr}:</strong><br /><br /><ul>${p.pending.map(x => `<li>${x}</li>`).join('')}</ul><br />Current readiness: <strong>${p.readiness}%</strong>.`;

      }


      return `I found the project in the SharePoint Project Register and checked simulated Planner/Procore status.` + statusCard(pr, p);

    }


    function sendMessage() {

      const text = input.value.trim();

      if (!text) return;

      addMessage('user', text);

      input.value = '';

      showTyping();

      setTimeout(() => {

        removeTyping();

        addMessage('bot', answer(text));

      }, 700);

    }


    function askQuick(text) {

      input.value = text;

      sendMessage();

    }
